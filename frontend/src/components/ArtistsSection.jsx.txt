import React, { useState, useEffect } from "react";
import axios from "axios";
import { Sparkles, Instagram, Award, Calendar, ArrowRight, ShieldCheck } from "lucide-react";
import { Button } from "./ui/button";

const BACKEND_URL = process.env.REACT_APP_BACKEND_URL || "";
const API = `${BACKEND_URL}/api`;

export default function ArtistsSection({ onSelectArtistForBooking }) {
  const [artists, setArtists] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchArtists = async () => {
      try {
        const res = await axios.get(`${API}/artists`);
        setArtists(Array.isArray(res.data) ? res.data : (res.data.artists || []));
      } catch (e) {
        console.error(e);
      } finally {
        setLoading(false);
      }
    };
    fetchArtists();
  }, []);

  return (
    <section id="artists" className="py-20 bg-[#0E0E10] border-t border-zinc-800">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        
        {/* Header */}
        <div className="text-left space-y-3 mb-12">
          <div className="inline-flex items-center gap-1.5 text-xs uppercase tracking-widest text-[#D4AF37] font-mono">
            <Sparkles className="w-3.5 h-3.5" /> Resident Guild
          </div>
          <h2 className="text-3xl sm:text-4xl font-serif font-black text-white">
            MEET THE MASTER ARTISTS & BARBERS
          </h2>
          <p className="text-sm sm:text-base text-zinc-400 max-w-2xl leading-relaxed">
            Each resident specializes in distinct stylistic disciplines—from monumental dark surrealism to razor-fade beard architecture.
          </p>
        </div>

        {/* Grid */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
          {artists.map((artist) => (
            <div
              key={artist.id}
              data-testid={`artist-profile-card-${artist.id}`}
              className="group bg-[#141416] border border-zinc-800 rounded-2xl overflow-hidden hover:border-[#D4AF37]/50 transition-all duration-300 flex flex-col justify-between text-left shadow-xl"
            >
              <div>
                {/* Image & Experience Badge */}
                <div className="relative aspect-[3/4] overflow-hidden bg-zinc-950">
                  <img
                    src={artist.avatar_url}
                    alt={artist.name}
                    className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                  />
                  <div className="absolute inset-0 bg-gradient-to-t from-[#141416] via-transparent to-transparent" />
                  
                  <div className="absolute top-3 right-3 bg-black/80 backdrop-blur-md px-2.5 py-1 rounded-md border border-white/10 text-[11px] font-mono text-[#D4AF37]">
                    ★ {artist.rating} ({artist.experience})
                  </div>
                </div>

                {/* Info */}
                <div className="p-5 space-y-3 -mt-6 relative z-10">
                  <div>
                    <h3 className="font-serif font-bold text-lg text-white group-hover:text-[#D4AF37] transition-colors">
                      {artist.name}
                    </h3>
                    <p className="text-xs text-[#D4AF37] font-medium">{artist.role}</p>
                    <p className="text-[11px] text-zinc-500 font-mono mt-0.5">{artist.instagram}</p>
                  </div>

                  <p className="text-xs text-zinc-400 line-clamp-3 leading-relaxed">
                    {artist.bio}
                  </p>

                  <div className="flex flex-wrap gap-1.5 pt-1">
                    {artist.specialties?.map((spec) => (
                      <span key={spec} className="text-[10px] px-2 py-0.5 rounded bg-zinc-900 border border-zinc-800 text-zinc-300">
                        {spec}
                      </span>
                    ))}
                  </div>
                </div>
              </div>

              {/* Action */}
              <div className="p-5 pt-0">
                <Button
                  data-testid={`book-with-artist-btn-${artist.id}`}
                  onClick={() => onSelectArtistForBooking(artist)}
                  className="w-full bg-zinc-900 border border-zinc-700 hover:bg-[#D4AF37] hover:text-black hover:border-[#D4AF37] text-zinc-200 font-semibold text-xs py-2 transition-all"
                >
                  <Calendar className="w-3.5 h-3.5 mr-1.5" /> Book With {artist.name.split(" ")[0]}
                </Button>
              </div>
            </div>
          ))}
        </div>

      </div>
    </section>
  );
}
